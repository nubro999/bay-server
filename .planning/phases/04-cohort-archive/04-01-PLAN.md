```typescript
export async function toggleCohortArchive(id: string) {
  await verifySession()

  const cohort = await prisma.cohort.findUnique({ where: { id } })
  if (!cohort) {
    return { error: 'Cohort not found' }
  }

  await prisma.cohort.update({
    where: { id },
    data: { isActive: !cohort.isActive },
  })

  revalidatePath('/admin/cohorts')
  revalidatePath('/')
}
```

Key details:
- Uses verifySession() guard (same pattern as all admin actions)
- Toggles the boolean — no separate archive/unarchive actions needed
- Revalidates both admin cohorts page and public root page
- Does NOT use redirect — stays on the same admin cohorts page
- Import revalidatePath is already imported in the file
